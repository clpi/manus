# duo Roadmap Implementation Plan

> **For Hermes:** Use subagent-driven-development skill to implement this plan task-by-task.

**Goal:** Implement the four highest-priority roadmap items: escape analysis for stack allocation, ARC pruning on non-escaping locals, compile-time metaprogramming (quote/unquote/macros), and concept/metatable unification.

**Architecture:** Add an escape analysis pass between sema and arc that marks variables as escaping/escaping-to-heap. Consume those marks in the ARC pass to skip retain/release on non-escaping locals. Add AST quote/unquote expressions and macro_def statements, with a macro expansion pass before sema. Unify concepts and metatables by having concept satisfaction install metatables with metamethods.

**Tech Stack:** Zig 0.17.0-dev, zig build, clang (codegen target)

fn analyze_closure_upvalues(self: *Sema, fb: *ast.FuncBody) !void {
    var names = std.ArrayList([]const u8).empty;
    var flags = std.ArrayList(bool).empty;
    try collect_upvalue_names(fb, &fb.body, fb.params, &names, &flags, self);
    fb.upvalues = try self.alloc.alloc(ast.Upvalue, names.items.len);
    for (names.items, flags.items, 0..) |nm, is_local, i| {
        var typ: ?RT = null;
        if (is_local) {

                typ = sym.typ;
                sym.captured_by_closure = true;  // NEW: mark as escaping

            typ = g_typ;
        }
        fb.upvalues[i] = .{ .name = nm, .is_local = is_local, .typ = typ };

}

### Task 4: Write escape analysis query function

**Objective:** Create a function that determines if a local can be stack-allocated.

**Step 1: Write the analysis module**

```zig
const std = @import("std");
const Sema = @import("sema.zig").Sema;
const Symbol = @import("sema.zig").Symbol;
const RT = @import("types.zig").RT;

pub fn canStackAllocate(sym: *const Symbol) bool {
    if (sym.is_global) return false;
    if (sym.escapes) return false;
    if (sym.captured_by_closure) return false;
    if (sym.address_taken) return false;
    return true;
}

pub fn shouldPruneArc(sym: *const Symbol) bool {
    // If a variable doesn't escape and isn't captured, we can skip retain/release
    if (sym.is_global) return false;
    if (sym.escapes) return false;
    if (sym.captured_by_closure) return false;
    return true;
}
```

fn popScope(self: *ArcPass) !void {
    const scope = self.scopes.pop();
    var i = scope.items.len;
    while (i > 0) {
        i -= 1;
        const tracked = scope.items[i];
        // NEW: skip release for non-escaping locals
        if (self.escape_info.get(tracked.name)) |sym| {
            if (escape.shouldPruneArc(sym)) continue;

        if (tracked.is_close) {
            try self.annotations.append(...{ .op = .close, ... });
        }
        try self.annotations.append(...{ .op = .release, ... });
    }
}

// macro name(param1, param2) <body>
fn parseMacroDef(self: *Parser) ParseError!*ast.Stmt {
    try self.expect(.macro);
    const name = try self.expectIdent();
    try self.expect(.lparen);
    var params = std.ArrayList([]const u8).empty;
    while (!self.check(.rparen)) {
        try params.append(try self.expectIdent());
        if (!self.check(.rparen)) try self.expect(.comma);
    }
    try self.expect(.rparen);
    const body = try self.parseExpr(); // should be a quote expression
    return try self.alloc.create(ast.Stmt{ .macro_def = .{
        .loc = self.prevLoc(),
        .name = name,
        .params = try params.toOwnedSlice(),
        .body = body,
    } });
}
